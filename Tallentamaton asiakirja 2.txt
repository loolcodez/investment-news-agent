def make_fingerprint(item):
    if item.url:
        return sha256(canonicalize_url(item.url)).hexdigest()
    return sha256(normalize_title(item.title) + item.source).hexdigest()


new_items = []

for item in fetched_items:
    fp = make_fingerprint(item)

    if memory.has_seen(fp):
        memory.update_last_seen(fp)
        continue

    memory.mark_seen(fp, item)
    new_items.append(item)
